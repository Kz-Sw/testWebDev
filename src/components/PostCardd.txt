---
import { toKanjiDigits, slugify } from "../scripts/utils";
import { categoryConfig } from "../scripts/categories";

const { post } = Astro.props;
const { 
  heroTitle, 
  author, 
  number,        // Using the new universal 'number' 
  category, 
  heroImage, 
  tags, 
  enTitle 
} = post.data;

const imageUrl =
  typeof heroImage === "string"
    ? heroImage
    : heroImage?.url;

const imageAlt =
  typeof heroImage === "string"
    ? heroTitle
    : heroImage?.alt || heroTitle;

const imagePosition =
  typeof heroImage === "string"
    ? "center center"
    : heroImage?.position || "center center";

// 1. Generate URL: /[category]/[000-slugified-en-title]
// Logic matches [slug].astro to ensure links work [cite: 10]
const fileName = post.file?.split('/').pop()?.replace('.mdx', '') || '';
const slugBase = enTitle ? slugify(enTitle) : fileName;
const paddedNum = number ? String(number).padStart(3, '0') : null;
const actualSlug = paddedNum ? `${paddedNum}-${slugBase}` : slugBase;

// Path is now directly under category (no /posts/ prefix) 
const postUrl = `/${category}/${actualSlug}`;

// 2. Generate Full Title using modular categoryConfig [cite: 13, 14]
const config = categoryConfig[category];
const kanjiNum = number ? toKanjiDigits(number) : "";
const fullTitle = config 
  ? `${config.prefix || "第"}${kanjiNum}${config.suffix}　${heroTitle}`
  : heroTitle;
---

<article class="blog-card" data-tags={tags?.join(',')}>
  <a href={postUrl} class="card-link">
    <div class="card-image">
      {heroImage && (
        <img 
          src={imageUrl}
          alt={imageAlt}
          style={`object-position: ${imagePosition};`}
          loading="lazy"
        />
      )}
      <div class="image-overlay"></div>

      <div class="card-main-overlay"> 
        <div class="card-title-section-overlay">
          <h2 class="card-title vertical-text">{fullTitle}</h2>
        </div>
        
        {author && (
          <p class="card-author vertical-text">{author}</p>
        )}
      </div>
    </div>
  </a>
</article>
